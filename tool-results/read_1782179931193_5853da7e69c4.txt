    44→function SettingsContent({ panelType, onClose }: { panelType: string; onClose: () => void }) {
    45→  switch (panelType) {
    46→    case "settings_account": return <AccountSettings onClose={onClose} />;
    47→    case "settings_privacy": return <PrivacySettings onClose={onClose} />;
    48→    case "settings_notifications": return <NotificationsSettings onClose={onClose} />;
    49→    case "settings_chats": return <ChatsSettings onClose={onClose} />;
    50→    case "settings_storage": return <StorageSettings onClose={onClose} />;
    51→    case "settings_appearance": return <AppearanceSettings onClose={onClose} />;
    52→    case "settings_accessibility": return <AccessibilitySettings onClose={onClose} />;
    53→    case "settings_help": return <HelpSettings onClose={onClose} />;
    54→    case "settings_linked_devices": return <LinkedDevicesSettings onClose={onClose} />;
    55→    case "settings_blocked": return <BlockedSettings onClose={onClose} />;

    61→// ==================== ACCOUNT SETTINGS ====================
    62→function AccountSettings({ onClose }: { onClose: () => void }) {
    63→  const { user, updateProfile } = useAuthStore();
    64→  const [editing, setEditing] = useState(false);
    65→  const [name, setName] = useState(user?.displayName || "");
    66→  const [about, setAbout] = useState(user?.about || "");
    67→
    68→  function save() {
    69→    updateProfile({ displayName: name, about });
    70→    setEditing(false);
    71→  }
    72→
    73→  if (!user) return null;
    74→
    75→  return (
    76→    <>
    77→      <SettingsHeader title="Account" onClose={onClose} />
    78→      <div className="flex-1 overflow-y-auto neu-scroll px-4 pb-6">
    79→        {/* Profile card */}
    80→        <div className="neu-raised rounded-3xl p-5 mb-4">
    81→          <div className="flex items-center gap-4">
    82→            <NeuAvatar initial={user.avatarInitial} gradient={user.avatarColor} size={72} />
    83→            <div className="flex-1">
    84→              <div className="text-xs neu-text-muted">Display name</div>
    85→              {editing ? (
    86→                <NeuInput value={name} onChange={(e) => setName(e.target.value)} containerClassName="mt-1" />
    87→              ) : (
    88→                <div className="font-semibold neu-text">{user.displayName}</div>
    89→              )}
    90→            </div>
    91→            <button onClick={() => editing ? save() : setEditing(true)} className="neu-accent rounded-xl px-3 py-2 text-xs text-white">
    92→              {editing ? "Save" : "Edit"}
    93→            </button>
    94→          </div>
    95→        </div>
    96→
    97→        {/* Zaxo Number */}
    98→        <Section title="Zaxo Number">
    99→          <div className="neu-raised rounded-2xl p-4 mb-3">
   100→            <div className="text-xs neu-text-muted uppercase tracking-wider">Your permanent Zaxo number</div>
   101→            <div className="text-3xl font-bold neu-text-accent font-mono mt-2 tracking-wider">{user.zaxoNumber}</div>
   102→            <div className="text-xs neu-text-muted mt-2">
   103→              This is your permanent identity on Zaxo. It cannot be changed.
   104→            </div>
   105→          </div>
   106→          <NeuSettingRow icon={<UserIcon size={18} />} label="Copy Zaxo number" onClick={() => navigator.clipboard?.writeText(user.zaxoNumber)} />
   107→        </Section>
   108→
   109→        {/* About */}
   110→        <Section title="About">
   111→          <div className="neu-raised rounded-2xl p-4 mb-3">
   112→            <div className="text-xs neu-text-muted mb-2">About</div>
   113→            {editing ? (
   114→              <NeuInput value={about} onChange={(e) => setAbout(e.target.value)} />
   115→            ) : (
   116→              <div className="text-sm neu-text">{user.about}</div>
   117→            )}
   118→            <div className="text-xs neu-text-muted mt-3">Email: {user.email}</div>
   119→          </div>
   120→        </Section>
   121→
   122→        <Section title="Account Actions">
   123→          <NeuSettingRow icon={<Shield size={18} />} label="Request account info" description="Download your data" showChevron />
   124→          <NeuSettingRow icon={<Trash2 size={18} />} label="Delete account" description="Permanently delete (30-day grace)" danger showChevron onClick={() => {
   125→            if (confirm("Delete account? This starts a 30-day grace period.")) {
   126→              useAuthStore.getState().signOut();
   127→              window.location.reload();
   128→            }
   129→          }} />
   130→        </Section>
   131→      </div>
   132→    </>
   133→  );
   134→}

   577→// ==================== TWO STEP VERIFICATION ====================
   578→function TwoStepSettings({ onClose }: { onClose: () => void }) {
   579→  const { settings, updateSettings } = useSettingsStore();
   580→  const [pin, setPin] = useState("");
   581→  const [enabled, setEnabled] = useState(settings.twoStepVerificationEnabled);
   582→
   583→  function enable() {
   584→    if (pin.length < 6) {
   585→      alert("PIN must be at least 6 digits");
   586→      return;
   587→    }
   588→    updateSettings({ twoStepVerificationEnabled: true });
   589→    setEnabled(true);
   590→  }
   591→
   592→  function disable() {
   593→    updateSettings({ twoStepVerificationEnabled: false });
   594→    setEnabled(false);
   595→    setPin("");
   596→  }
   597→
   598→  return (
   599→    <>
   600→      <SettingsHeader title="Two-Step Verification" onClose={onClose} />
   601→      <div className="flex-1 overflow-y-auto neu-scroll px-4 pb-6">
   602→        <Section title="Status">
   603→          <div className="neu-raised rounded-2xl p-4 mb-3">
   604→            <div className="text-xs neu-text-muted">Two-step verification</div>
   605→            <div className="text-base font-semibold neu-text mt-1">{enabled ? "Enabled" : "Disabled"}</div>
   606→            <div className="text-xs neu-text-muted mt-2">
   607→              {enabled
   608→                ? "Your account is protected with a PIN in addition to your password."
   609→                : "Add an extra layer of security. You'll need a PIN to register your phone number with Zaxo again."}
   610→            </div>
   611→          </div>
   612→        </Section>
   613→
   614→        {!enabled ? (
   615→          <Section title="Set up PIN">
   616→            <NeuInput
   617→              type="password"
   618→              placeholder="Enter 6-digit PIN"
   619→              value={pin}
   620→              onChange={(e) => setPin(e.target.value.replace(/\D/g, "").slice(0, 6))}
   621→              icon={<Shield size={16} />}
   622→            />
   623→            <NeuButton variant="accent" fullWidth rounded="xl" className="mt-3" onClick={enable}>Enable</NeuButton>
   624→          </Section>
   625→        ) : (
   626→          <>
   627→            <Section title="PIN management">
   628→              <NeuSettingRow label="Change PIN" showChevron />
   629→              <NeuSettingRow label="Recovery email" description="Not set" showChevron />
   630→            </Section>
   631→            <NeuButton variant="danger" fullWidth rounded="xl" icon={<Trash2 size={16} />} onClick={disable}>Disable</NeuButton>
   632→          </>
   633→        )}
   634→      </div>
   635→    </>
   636→  );
   637→}

   649→function ToggleRow({ icon, label, description, checked, onChange }: {
