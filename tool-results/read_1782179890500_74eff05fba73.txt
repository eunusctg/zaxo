    21→
    22→export function ChatRoom({ chatId }: { chatId: string }) {
    23→  const { chats, contacts, messages, sendMessage, markChatRead, setTyping, toggleReaction } = useAppStore();
    24→  const { setSubPanel } = useUIStore();
    25→  const { user } = useAuthStore();
    26→  const [draft, setDraft] = useState("");
    27→  const [showEmoji, setShowEmoji] = useState(false);
    28→  const [showAttach, setShowAttach] = useState(false);
    29→  const [recording, setRecording] = useState(false);
    30→  const [recordSeconds, setRecordSeconds] = useState(0);
    31→  const [replyTo, setReplyTo] = useState<Message | null>(null);
    32→  const [activeMessage, setActiveMessage] = useState<Message | null>(null);
    33→  const [showReactionPicker, setShowReactionPicker] = useState<string | null>(null);
    34→
    35→  const chat = chats.find((c) => c.id === chatId);
    36→  const chatMessages = messages[chatId] || [];
    37→  const scrollRef = useRef<HTMLDivElement>(null);
    38→
    39→  // Mark as read on open
    40→  useEffect(() => {
    41→    markChatRead(chatId);
    42→  }, [chatId, markChatRead]);
    43→
    44→  // Simulate other user typing then responding
    45→  const otherTyping = chat?.typingUserIds.length || 0;
    46→
    47→  // Auto-scroll to bottom
    48→  useEffect(() => {
    49→    if (scrollRef.current) {
    50→      scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
    51→    }
    52→  }, [chatMessages.length, otherTyping]);
    53→
    54→  // Recording timer
    55→  useEffect(() => {
    56→    if (recording) {
    57→      const interval = setInterval(() => setRecordSeconds((s) => s + 1), 1000);
    58→      return () => clearInterval(interval);
    59→    } else {
    60→      setRecordSeconds(0);
    61→    }
    62→  }, [recording]);
    63→
    64→  if (!chat) {
    65→    return (
    66→      <div className="flex items-center justify-center h-full neu-text-muted">
    67→        Chat not found
    68→      </div>
    69→    );
    70→  }
    71→
    72→  const other = chat.type === "individual" ? getContactById(chat.participantIds[0], contacts) : null;
    73→  const name = chat.type === "group" ? chat.name || "Group" : other?.displayName || "Unknown";
    74→  const subtitle = chat.type === "group"
    75→    ? `${chat.participantIds.length + 1} members`
    76→    : other?.online ? "online" : other ? `last seen ${relativeShort(other.lastSeen)}` : "";
    77→  const avatarInitial = chat.type === "group" ? chat.avatarInitial || "G" : other?.avatarInitial || "?";
    78→  const avatarGradient = chat.type === "group" ? chat.avatarColor || "" : other?.avatarColor || "";
    79→
    80→  function handleSend() {
    81→    if (!draft.trim()) return;
    82→    sendMessage(chatId, draft.trim());
    83→    setDraft("");
    84→    setReplyTo(null);
    85→    // Simulate a reply for individual chats
    86→    if (chat?.type === "individual") {
    87→      const otherId = chat.participantIds[0];
    88→      setTimeout(() => {
    89→        setTyping(chatId, [otherId]);
    90→        setTimeout(() => {
    91→          setTyping(chatId, []);
    92→          const replies = ["Got it! 👍", "Haha nice", "Sounds good", "Talk soon!", "👀", "Will do!", "Perfect!"];
    93→          const reply = replies[Math.floor(Math.random() * replies.length)];
    94→          // Use store directly to send from other user
    95→          useAppStore.setState((state) => {
    96→            const newMsg: Message = {
    97→              id: `m_${Date.now()}_other`,
    98→              chatId,
    99→              senderId: otherId,
   100→              type: "text",
   101→              text: reply,
   102→              timestamp: Date.now(),
   103→              status: "delivered",
   104→              reactions: {},
   105→              disappearing: "off",
   106→            };
   107→            return {
   108→              messages: {
   109→                ...state.messages,
   110→                [chatId]: [...(state.messages[chatId] || []), newMsg],
   111→              },
   112→              chats: state.chats.map((c) =>
   113→                c.id === chatId ? { ...c, lastMessage: newMsg, updatedAt: newMsg.timestamp } : c,
   114→              ),
   115→            };
   116→          });
   117→        }, 1500);
   118→      }, 800);
   119→    }
   120→  }
   121→
   122→  function handleVoiceSend() {
   123→    if (recordSeconds < 1) return;
   124→    sendMessage(chatId, "", "voice", { duration: recordSeconds });
   125→    setRecording(false);
   126→  }
   127→
   128→  // Group messages by day
   129→  const grouped = (() => {
   130→    const groups: { date: string; items: Message[] }[] = [];
   131→    chatMessages.forEach((m) => {
   132→      const d = formatDateSeparator(m.timestamp);
   133→      const last = groups[groups.length - 1];
   134→      if (last && last.date === d) {
   135→        last.items.push(m);
   136→      } else {
   137→        groups.push({ date: d, items: [m] });
   138→      }
   139→    });
   140→    return groups;
   141→  })();
   142→
   143→  return (
   144→    <div className="flex flex-col h-full" style={{ background: "var(--neu-bg)" }}>
   145→      {/* Header */}
   146→      <header className="px-3 pt-4 pb-3 flex items-center gap-2 neu-raised-sm rounded-b-3xl">
   147→        <button onClick={() => setSubPanel({ type: "none" })} className="neu-text p-1">
   148→          <ChevronLeft size={24} />
   149→        </button>
   150→        <button
   151→          onClick={() => setSubPanel({ type: chat.type === "group" ? "group_info" : "chat_info", chatId })}
   152→          className="flex items-center gap-3 flex-1 min-w-0"
   153→        >
   154→          <NeuAvatar initial={avatarInitial} gradient={avatarGradient} size={40} online={other?.online} />
   155→          <div className="text-left min-w-0">
   156→            <div className="font-semibold text-sm neu-text truncate">{name}</div>
   157→            <div className="text-xs neu-text-muted truncate">
   158→              {otherTyping > 0 ? (
   159→                <span className="neu-text-accent">typing…</span>
   160→              ) : subtitle}
   161→            </div>
   162→          </div>
   163→        </button>
   164→        <button onClick={() => setSubPanel({ type: "call_screen", otherUserId: chat.type === "individual" ? chat.participantIds[0] : "", callType: "voice" })} className="neu-pressable rounded-full w-10 h-10 flex items-center justify-center neu-text">
   165→          <Phone size={18} />
   166→        </button>
   167→        <button onClick={() => setSubPanel({ type: "call_screen", otherUserId: chat.type === "individual" ? chat.participantIds[0] : "", callType: "video" })} className="neu-pressable rounded-full w-10 h-10 flex items-center justify-center neu-text">
   168→          <Video size={18} />
   169→        </button>
   170→        <button className="neu-pressable rounded-full w-10 h-10 flex items-center justify-center neu-text">
   171→          <MoreVertical size={18} />
   172→        </button>
   173→      </header>
   174→
   175→      {/* Messages */}
   176→      <div ref={scrollRef} className="flex-1 overflow-y-auto neu-scroll px-3 py-4">
   177→        {chatMessages.length === 0 && (
   178→          <div className="flex flex-col items-center justify-center h-full text-center px-8">
   179→            <div className="neu-raised-sm rounded-2xl px-5 py-4 text-xs neu-text-muted max-w-xs">
   180→              🔒 Messages are end-to-end encrypted. No one outside this chat can read them.
   181→            </div>
   182→          </div>
   183→        )}
   184→        {grouped.map((group) => (
   185→          <div key={group.date}>
   186→            <div className="flex justify-center my-3">
   187→              <div className="neu-raised-sm rounded-full px-3 py-1 text-[10px] neu-text-muted font-medium">
   188→                {group.date}
   189→              </div>
   190→            </div>
   191→            {group.items.map((msg, i) => {
   192→              const isMine = msg.senderId === "me";
   193→              const prev = group.items[i - 1];
   194→              const showAvatar = chat.type === "group" && !isMine && (!prev || prev.senderId !== msg.senderId);
   195→              return (
   196→                <MessageBubble
   197→                  key={msg.id}
   198→                  msg={msg}
   199→                  isMine={isMine}
   200→                  chat={chat}
   201→                  showAvatar={showAvatar}
   202→                  onReply={() => setReplyTo(msg)}
   203→                  onReact={() => setShowReactionPicker(showReactionPicker === msg.id ? null : msg.id)}
   204→                  onLongPress={() => setActiveMessage(msg)}
   205→                  reactionPickerOpen={showReactionPicker === msg.id}
   206→                  onReactionPick={(emoji) => {
   207→                    toggleReaction(chatId, msg.id, emoji, user?.id || "me");
   208→                    setShowReactionPicker(null);
   209→                  }}
   210→                />
   211→              );
   212→            })}
   213→          </div>
   214→        ))}
   215→        {otherTyping > 0 && (
   216→          <div className="flex items-center gap-2 mt-1 mb-3">
   217→            {chat.type === "group" && <NeuAvatar initial={other?.avatarInitial || "?"} gradient={other?.avatarColor || ""} size={28} />}
   218→            <div className="neu-raised-sm rounded-2xl rounded-bl-md px-4 py-3 flex items-center gap-1">
   219→              <span className="typing-dot w-1.5 h-1.5 rounded-full neu-text-muted" />
   220→              <span className="typing-dot w-1.5 h-1.5 rounded-full neu-text-muted" />
   221→              <span className="typing-dot w-1.5 h-1.5 rounded-full neu-text-muted" />
   222→            </div>
   223→          </div>
   224→        )}
   225→        <div className="h-2" />
   226→      </div>
   227→
   228→      {/* Reply preview */}
   229→      <AnimatePresence>
   230→        {replyTo && (
   231→          <motion.div
   232→            initial={{ height: 0, opacity: 0 }}
   233→            animate={{ height: "auto", opacity: 1 }}
   234→            exit={{ height: 0, opacity: 0 }}
   235→            className="px-3 overflow-hidden"
   236→          >
   237→            <div className="neu-inset rounded-xl px-3 py-2 flex items-center gap-2">
   238→              <Reply size={14} className="neu-text-accent" />
   239→              <div className="flex-1 min-w-0">
   240→                <div className="text-xs font-medium neu-text-accent">
   241→                  Replying to {replyTo.senderId === "me" ? "yourself" : name.split(" ")[0]}
   242→                </div>
   243→                <div className="text-xs neu-text-muted truncate">
   244→                  {replyTo.text || `[${replyTo.type}]`}
   245→                </div>
   246→              </div>
   247→              <button onClick={() => setReplyTo(null)} className="neu-text-muted">
   248→                <X size={16} />
   249→              </button>
   250→            </div>
   251→          </motion.div>
   252→        )}
   253→      </AnimatePresence>
   254→
   255→      {/* Attachment panel */}
   256→      <AnimatePresence>
   257→        {showAttach && (
   258→          <motion.div
   259→            initial={{ height: 0, opacity: 0 }}
   260→            animate={{ height: "auto", opacity: 1 }}
   261→            exit={{ height: 0, opacity: 0 }}
   262→            className="px-4 overflow-hidden"
   263→          >
   264→            <div className="grid grid-cols-4 gap-3 py-3">
   265→              <AttachOption icon={<Camera size={22} />} label="Camera" color="#FF6B6B" onClick={() => { sendMessage(chatId, "📸 Photo captured", "image", { mediaName: "photo.jpg" }); setShowAttach(false); }} />
   266→              <AttachOption icon={<FileText size={22} />} label="Document" color="#6C5CE7" onClick={() => { sendMessage(chatId, "", "document", { mediaName: "document.pdf", mediaSize: 245000 }); setShowAttach(false); }} />
   267→              <AttachOption icon={<MapPin size={22} />} label="Location" color="#00B894" onClick={() => { sendMessage(chatId, "", "location", { location: { lat: 3.139, lng: 101.6869, label: "Kuala Lumpur" } }); setShowAttach(false); }} />
   268→              <AttachOption icon={<User size={22} />} label="Contact" color="#FECA57" onClick={() => { sendMessage(chatId, "", "contact", { contactCard: { name: "Alex Rivera", zaxoNumber: "246-810-935" } }); setShowAttach(false); }} />
   269→            </div>
   270→          </motion.div>
   271→        )}
   272→      </AnimatePresence>
   273→
   274→      {/* Input bar */}
   275→      <div className="px-3 pb-3 pt-2">
   276→        {recording ? (
   277→          <div className="neu-raised-sm rounded-2xl px-4 py-3 flex items-center gap-3">
   278→            <div className="w-3 h-3 rounded-full" style={{ background: "var(--neu-danger)" }} />
   279→            <span className="text-sm neu-text font-mono">
   280→              {Math.floor(recordSeconds / 60)}:{(recordSeconds % 60).toString().padStart(2, "0")}
   281→            </span>
   282→            <div className="flex-1 h-6 flex items-center">
   283→              {/* Fake waveform */}
   284→              {Array.from({ length: 30 }).map((_, i) => (
   285→                <span
   286→                  key={i}
   287→                  className="waveform-bar"
   288→                  style={{ height: `${20 + Math.sin(i * 1.7 + recordSeconds) * 12 + Math.random() * 8}%` }}
   289→                />
   290→              ))}
   291→            </div>
   292→            <button onClick={() => setRecording(false)} className="neu-text-muted">
   293→              <X size={20} />
   294→            </button>
   295→            <button onClick={handleVoiceSend} className="neu-accent rounded-full w-10 h-10 flex items-center justify-center">
   296→              <Send size={16} />
   297→            </button>
   298→          </div>
   299→        ) : (
   300→          <div className="flex items-end gap-2">
   301→            <div className="flex-1 neu-raised-sm rounded-2xl px-3 py-2 flex items-end gap-2">
   302→              <button
   303→                onClick={() => { setShowEmoji(!showEmoji); setShowAttach(false); }}
   304→                className="neu-text-muted hover:neu-text-accent p-1.5 shrink-0"
   305→              >
   306→                <Smile size={20} />
   307→              </button>
   308→              <textarea
   309→                value={draft}
   310→                onChange={(e) => setDraft(e.target.value)}
   311→                onKeyDown={(e) => {
   312→                  if (e.key === "Enter" && !e.shiftKey) {
   313→                    e.preventDefault();
   314→                    handleSend();
   315→                  }
   316→                }}
   317→                placeholder="Message"
   318→                rows={1}
   319→                className="flex-1 bg-transparent outline-none border-none text-sm neu-text placeholder:text-[color:var(--neu-text-muted)] resize-none max-h-24 py-1.5"
   320→                style={{ minHeight: 24 }}
   321→              />
   322→              <button
   323→                onClick={() => { setShowAttach(!showAttach); setShowEmoji(false); }}
   324→                className="neu-text-muted hover:neu-text-accent p-1.5 shrink-0"
   325→              >
   326→                <Paperclip size={20} />
   327→              </button>
   328→              <button className="neu-text-muted hover:neu-text-accent p-1.5 shrink-0">
   329→                <Camera size={20} />
   330→              </button>
   331→            </div>
   332→            <button
   333→              onClick={() => { if (draft.trim()) handleSend(); else setRecording(true); }}
   334→              className="neu-accent rounded-full w-12 h-12 flex items-center justify-center shrink-0"
   335→              aria-label={draft.trim() ? "Send" : "Record voice message"}
   336→            >
   337→              {draft.trim() ? <Send size={20} /> : <Mic size={20} />}
   338→            </button>
   339→          </div>
   340→        )}
   341→
   342→        {/* Quick emoji */}
   343→        <AnimatePresence>
   344→          {showEmoji && (
   345→            <motion.div
   346→              initial={{ opacity: 0, y: 8 }}
   347→              animate={{ opacity: 1, y: 0 }}
   348→              exit={{ opacity: 0, y: 8 }}
   349→              className="mt-2 neu-raised-sm rounded-2xl p-3 grid grid-cols-8 gap-1"
   350→            >
   351→              {["😀","😂","😍","🥰","😘","😎","🤔","😴","🙄","😡","😭","🥺","😱","🤯","🤩","😈","👍","👎","👏","🙌","🙏","💪","🔥","✨","❤️","💔","💯","🎉","🎂","🍕","☕","🌈"].map((e) => (
   352→                <button
   353→                  key={e}
   354→                  onClick={() => setDraft(draft + e)}
   355→                  className="text-xl p-1.5 rounded-lg hover:neu-inset transition-all"
   356→                >
   357→                  {e}
   358→                </button>
   359→              ))}
   360→            </motion.div>
   361→          )}
   362→        </AnimatePresence>
   363→      </div>
   364→
   365→      {/* Message action sheet */}
   366→      <AnimatePresence>
   367→        {activeMessage && (
   368→          <MessageActionSheet
   369→            msg={activeMessage}
   370→            chat={chat}
   371→            onClose={() => setActiveMessage(null)}
   372→            onReply={() => { setReplyTo(activeMessage); setActiveMessage(null); }}
   373→          />
   374→        )}
   375→      </AnimatePresence>
   376→    </div>
   377→  );
   378→}
   379→
   380→function MessageBubble({
   381→  msg, isMine, chat, showAvatar, onReply, onReact, onLongPress, reactionPickerOpen, onReactionPick,
   382→}: {
   383→  msg: Message;
   384→  isMine: boolean;
   385→  chat: Chat;
   386→  showAvatar: boolean;
   387→  onReply: () => void;
   388→  onReact: () => void;
   389→  onLongPress: () => void;
   390→  reactionPickerOpen: boolean;
   391→  onReactionPick: (e: string) => void;
   392→}) {
   393→  const { contacts } = useAppStore();
   394→  const sender = msg.senderId === "me" ? null : getContactById(msg.senderId, contacts);
   395→  const longPressTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   396→
   397→  function handlePressStart() {
   398→    longPressTimer.current = setTimeout(() => onLongPress(), 500);
   399→  }
   400→  function handlePressEnd() {
   401→    if (longPressTimer.current) clearTimeout(longPressTimer.current);
   402→  }
   403→
   404→  if (msg.type === "system") {
   405→    return (
   406→      <div className="flex justify-center my-2">
   407→        <div className="neu-raised-sm rounded-full px-3 py-1 text-[11px] neu-text-muted">
   408→          {msg.text}
   409→        </div>
   410→      </div>
   411→    );
   412→  }
   413→
   414→  const reactions = Object.entries(msg.reactions);
   415→  const reactionGroups: { [emoji: string]: number } = {};
   416→  reactions.forEach(([, emoji]) => {
   417→    reactionGroups[emoji] = (reactionGroups[emoji] || 0) + 1;
   418→  });
   419→
   420→  return (
   421→    <div className={`flex items-end gap-2 mb-1.5 ${isMine ? "justify-end" : "justify-start"}`}>
   422→      {showAvatar && sender && (
   423→        <NeuAvatar initial={sender.avatarInitial} gradient={sender.avatarColor} size={28} />
   424→      )}
   425→      {!showAvatar && !isMine && <div style={{ width: 28 }} />}
   426→      <div className={`relative max-w-[78%] ${isMine ? "items-end" : "items-start"}`}>
   427→        {showAvatar && chat.type === "group" && sender && (
   428→          <div className="text-[11px] font-medium neu-text-accent mb-0.5 ml-1">
   429→            {sender.displayName}
   430→          </div>
   431→        )}
   432→        <div
   433→          onPointerDown={handlePressStart}
   434→          onPointerUp={handlePressEnd}
   435→          onPointerLeave={handlePressEnd}
   436→          onClick={() => reactionPickerOpen && onReact()}
   437→          className={`px-3.5 py-2 text-sm cursor-pointer transition-all active:scale-[0.98] ${
   438→            isMine
   439→              ? "neu-accent text-white"
   440→              : "neu-raised-sm neu-text"
   441→          }`}
   442→          style={{
   443→            borderRadius: isMine
   444→              ? "18px 18px 4px 18px"
   445→              : "18px 18px 18px 4px",
   446→          }}
   447→        >
   448→          {msg.deletedForEveryone ? (
   449→            <span className="italic opacity-70 text-xs">🚫 This message was deleted</span>
   450→          ) : (
   451→            <MessageContent msg={msg} isMine={isMine} />
   452→          )}
   453→          <div className={`flex items-center gap-1 mt-1 ${isMine ? "justify-end" : "justify-start"}`}>
   454→            <span className={`text-[10px] ${isMine ? "text-white/70" : "neu-text-muted"}`}>
   455→              {formatTime(msg.timestamp)}
   456→              {msg.edited && <span className="ml-1">· edited</span>}
   457→            </span>
   458→            {isMine && !msg.deletedForEveryone && <MessageStatus status={msg.status} mine />}
   459→          </div>
   460→        </div>
   461→
   462→        {/* Reaction picker */}
   463→        {reactionPickerOpen && (
   464→          <motion.div
   465→            initial={{ opacity: 0, y: 4, scale: 0.95 }}
   466→            animate={{ opacity: 1, y: 0, scale: 1 }}
   467→            className={`absolute z-30 ${isMine ? "right-0" : "left-8"} -top-12 neu-raised rounded-full px-2 py-1.5 flex gap-1`}
   468→          >
   469→            {QUICK_EMOJIS.map((e) => (
   470→              <button
   471→                key={e}
   472→                onClick={(ev) => { ev.stopPropagation(); onReactionPick(e); }}
   473→                className="text-xl hover:scale-125 transition-transform p-1"
   474→              >
   475→                {e}
   476→              </button>
   477→            ))}
   478→          </motion.div>
   479→        )}
   480→
   481→        {/* Reactions */}
   482→        {Object.keys(reactionGroups).length > 0 && (
   483→          <div className={`flex gap-1 mt-1 ${isMine ? "justify-end" : "justify-start"}`}>
   484→            {Object.entries(reactionGroups).map(([emoji, count]) => (

   590→function VoiceMessagePlayer({ duration, isMine }: { duration: number; isMine: boolean }) {
   591→  const [playing, setPlaying] = useState(false);
   592→  const [progress, setProgress] = useState(0);
   593→
   594→  useEffect(() => {
   595→    if (playing) {
   596→      const interval = setInterval(() => {
   597→        setProgress((p) => {
   598→          if (p >= 100) {
   599→            setPlaying(false);
   600→            return 0;
   601→          }
   602→          return p + (100 / (duration * 10));
   603→        });
   604→      }, 100);
   605→      return () => clearInterval(interval);
   606→    }
   607→  }, [playing, duration]);
   608→
   609→  return (
   610→    <div className="flex items-center gap-2 py-1 pr-3 min-w-[180px]">
   611→      <button
   612→        onClick={(e) => { e.stopPropagation(); setPlaying(!playing); }}
   613→        className="w-9 h-9 rounded-full flex items-center justify-center shrink-0"
   614→        style={{ background: isMine ? "rgba(255,255,255,0.25)" : "var(--neu-accent)" }}
   615→      >
   616→        {playing ? <Pause size={16} className="text-white" /> : <Play size={16} className="text-white ml-0.5" />}
   617→      </button>
   618→      <div className="flex-1 h-7 flex items-center">
   619→        {Array.from({ length: 28 }).map((_, i) => {
   620→          const filled = (i / 28) * 100 < progress;
   621→          return (
   622→            <span
   623→              key={i}
   624→              className="waveform-bar"
   625→              style={{
   626→                height: `${20 + Math.sin(i * 1.4) * 14 + Math.random() * 10}%`,
   627→                background: filled ? (isMine ? "white" : "var(--neu-accent)") : (isMine ? "rgba(255,255,255,0.4)" : "var(--neu-text-muted)"),
   628→              }}
   629→            />
   630→          );
   631→        })}
   632→      </div>
   633→      <span className={`text-[11px] font-mono ${isMine ? "text-white/80" : "neu-text-muted"}`}>
   634→        {Math.floor(duration / 60)}:{(duration % 60).toString().padStart(2, "0")}
   635→      </span>
   636→    </div>
   637→  );
   638→}

   659→function MessageActionSheet({ msg, chat, onClose, onReply }: { msg: Message; chat: Chat; onClose: () => void; onReply: () => void }) {
   660→  const { toggleStarMessage, deleteMessage, editMessage } = useAppStore();
   661→  const [editing, setEditing] = useState(false);
   662→  const [editText, setEditText] = useState(msg.text || "");
   663→
   664→  const isMine = msg.senderId === "me";
   665→
   666→  function handleCopy() {
   667→    navigator.clipboard?.writeText(msg.text || "");
   668→    onClose();
   669→  }
   670→
   671→  function handleDelete(forEveryone: boolean) {
   672→    deleteMessage(chat.id, msg.id, forEveryone);
   673→    onClose();
   674→  }
   675→
   676→  function handleEdit() {
   677→    if (editText.trim() && editText !== msg.text) {
   678→      editMessage(chat.id, msg.id, editText.trim());
   679→    }
   680→    setEditing(false);
   681→    onClose();
   682→  }
   683→
   684→  return (
   685→    <motion.div
   686→      initial={{ opacity: 0 }}
   687→      animate={{ opacity: 1 }}
   688→      exit={{ opacity: 0 }}
   689→      className="fixed inset-0 z-50 flex items-end justify-center"
   690→      onClick={onClose}
   691→    >
   692→      <div className="absolute inset-0 bg-black/40" />
   693→      <motion.div
   694→        initial={{ y: 200 }}
   695→        animate={{ y: 0 }}
   696→        exit={{ y: 200 }}
   697→        transition={{ type: "spring", damping: 30 }}
   698→        onClick={(e) => e.stopPropagation()}
   699→        className="relative z-10 w-full max-w-md neu-raised rounded-t-3xl p-4 pb-6"
   700→      >
   701→        {/* Drag handle */}
   702→        <div className="w-12 h-1 rounded-full mx-auto neu-text-muted opacity-30 mb-3" />
   703→
   704→        {/* Message preview */}
   705→        {!editing && (
   706→          <div className="neu-inset rounded-2xl px-4 py-3 mb-4 max-h-32 overflow-y-auto neu-scroll">
   707→            <div className="text-xs neu-text-muted mb-1">
   708→              {formatDateSeparator(msg.timestamp)} · {formatTime(msg.timestamp)}
   709→            </div>
   710→            <div className="text-sm neu-text whitespace-pre-wrap break-words">
   711→              {msg.deletedForEveryone ? "🚫 This message was deleted" : msg.text || `[${msg.type}]`}
   712→            </div>
   713→          </div>
   714→        )}
   715→
   716→        {editing && (
   717→          <div className="mb-4">
   718→            <textarea
   719→              value={editText}
   720→              onChange={(e) => setEditText(e.target.value)}
   721→              className="neu-well rounded-2xl px-4 py-3 text-sm neu-text outline-none w-full resize-none"
   722→              rows={3}
   723→            />
   724→            <div className="flex gap-2 mt-2">
   725→              <NeuButton variant="raised" fullWidth rounded="xl" onClick={() => setEditing(false)}>Cancel</NeuButton>
   726→              <NeuButton variant="accent" fullWidth rounded="xl" onClick={handleEdit}>Save</NeuButton>
   727→            </div>
   728→          </div>
   729→        )}
   730→
   731→        {!editing && (
   732→          <div className="grid grid-cols-4 gap-3">
   733→            <ActionItem icon={<Reply size={18} />} label="Reply" onClick={onReply} />
   734→            <ActionItem icon={<Copy size={18} />} label="Copy" onClick={handleCopy} />
   735→            <ActionItem icon={<Star size={18} />} label={msg.starred ? "Unstar" : "Star"} onClick={() => { toggleStarMessage(chat.id, msg.id); onClose(); }} />
   736→            <ActionItem icon={<Forward size={18} />} label="Forward" onClick={onClose} />
   737→            {isMine && msg.type === "text" && (
   738→              <ActionItem icon={<Pencil size={18} />} label="Edit" onClick={() => setEditing(true)} />
   739→            )}
   740→            {isMine && (
   741→              <ActionItem icon={<Trash2 size={18} />} label="Delete" danger onClick={() => handleDelete(true)} />
   742→            )}
   743→          </div>
   744→        )}
   745→      </motion.div>
   746→    </motion.div>
   747→  );
   748→}

   764→function relativeShort(timestamp: number): string {
   765→  const diff = Date.now() - timestamp;
   766→  if (diff < 60000) return "just now";
   767→  if (diff < 3600000) return `${Math.floor(diff / 60000)} min ago`;
   768→  if (diff < 86400000) return `${Math.floor(diff / 3600000)}h ago`;
   769→  return `${Math.floor(diff / 86400000)}d ago`;
   770→}
